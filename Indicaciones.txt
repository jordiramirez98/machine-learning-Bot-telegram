# ===== Instalación y ejecución (Windows, CMD en VS Code) =====
# Requiere Python 3.11 (con 3.12+ las librerías dan problemas)

# 1. Crear y activar entorno virtual con Python 3.11
py -3.11 -m venv venv
venv\Scripts\activate

# 2. Instalar PyTorch CPU y luego el resto (versiones probadas que funcionan juntas)
python -m pip install --upgrade pip
pip install torch --index-url https://download.pytorch.org/whl/cpu
pip install -r requirements.txt

# 3. Probar que la IA funciona (la primera vez descarga el modelo, ~435 MB)
python -c "from pysentimiento import create_analyzer; a=create_analyzer(task='emotion', lang='es'); print(a.predict('te quiero mucho'))"
